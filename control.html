QUESTION 1:
Write a program to enter a number from the user. If the number is between 1 and 5, print the same number. Otherwise, print invalid number.

PROGRAM 1: Using Multiple If Statements

#include < stdio.h >

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    if (n == 1)
        printf("1");

    if (n == 2)
        printf("2");

    if (n == 3)
        printf("3");

    if (n == 4)
        printf("4");

    if (n == 5)
        printf("5");

    if (n < 1 || n > 5)
        printf("Invalid number");

    return 0;
}

OUTPUT:
Enter a number: 3
3


QUESTION 2:
Write a program to enter a number and check whether it is odd or even.

PROGRAM:

#include < stdio.h >

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    if (n % 2 == 0)
        printf("Number is even");
    else
        printf("Number is odd");

    return 0;
}

OUTPUT:
Enter a number: 10
Number is even

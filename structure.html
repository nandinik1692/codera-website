Question 1

Create a structure of student containing roll number, name, city and marks. Enter details from user and display them.

Solution:

#include < stdio.h >;

struct student
{
    int rollno;
    char name[30];
    char city[15];
    int marks;
};

int main()
{
    struct student st;

    printf("Enter Roll Number: ");
    scanf("%d", &st.rollno);

    printf("Enter the Name: ");
    scanf("%s", st.name);

    printf("Enter the City: ");
    scanf("%s", st.city);

    printf("Enter the Marks: ");
    scanf("%d", &st.marks);

    printf("Roll Number is: %d\n", st.rollno);
    printf("Name is: %s\n", st.name);
    printf("City is: %s\n", st.city);
    printf("Total Marks is: %d\n", st.marks);

    return 0;
}

Output:

Enter Roll Number: 101 Enter the Name: Nandini Enter the City: Chandigarh Enter the Marks: 95 Roll Number is: 101 Name is: Nandini City is: Chandigarh Total Marks is: 95

Question 2

Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form?

Solution:

#include < stdio.h >;
#include < conio.h >;

struct employee
{
    int id;
    char name[20];
    char designation[20];
};

void main()
{
    struct employee emp[50];
    int i;

    clrscr();

    for(i = 0; i < 50; i++)
    {
        printf("\nEnter ID: ");
        scanf("%d", &emp[i].id);

        printf("Enter Name: ");
        scanf("%s", emp[i].name);

        printf("Enter Designation: ");
        scanf("%s", emp[i].designation);
    }

    printf("\nID\tName\tDesignation");

    for(i = 0; i < 50; i++)
    {
        printf("\n%d\t%s\t%s",
        emp[i].id, emp[i].name, emp[i].designation);
    }
}

Output:

Enter ID: 101 Enter Name: Rahul Enter Designation: Manager Enter ID: 102 Enter Name: Aman Enter Designation: Clerk ID Name Designation 101 Rahul Manager 102 Aman Clerk

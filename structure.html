#include < stdio.h >

struct student
{
    int rollno;
    char name[30];
    char city[15];
    int marks;
};

int main()
{
    struct student st;

    printf("Enter Roll Number: ");
    scanf("%d", &st.rollno);

    printf("Enter the Name: ");
    scanf("%s", st.name);

    printf("Enter the City: ");
    scanf("%s", st.city);

    printf("Enter the Marks: ");
    scanf("%d", &st.marks);

    printf("Roll Number is: %d\n", st.rollno);
    printf("Name is: %s\n", st.name);
    printf("City is: %s\n", st.city);
    printf("Total Marks is: %d\n", st.marks);

    return 0;
}

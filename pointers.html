 Pointers in C Language

Question 1: Write a C program to display the value and address of a variable using a pointer.
Program:
 #include < stdio.h >

int main()
{
    int a = 50;
    int *p;

    p = &a;

    printf("Value is %d", a);
    printf("\nAddress is %p", (void *)&a);
    printf("\nValue using pointer is %d", *p);

    return 0;
}

output:
Value is 50
Address is 0x7ffe1234
Value using pointer is 50



Question 2: Write a program in which we are having a 2-d array of 2*3, insert the value thorugh pointer and print the value

Solution:
#include < stdio.h >

int main() {
    int arr[2][3];
    
    int *ptr = &arr[0][0];

    printf("Enter 6 numbers:\n");
    for (int i = 0; i < 6; i++) {
        scanf("%d", (ptr + i));
    }

    ptr = &arr[0][0];

    printf("\nThe 2D array elements are:\n");
    for (int i = 0; i < 2; i++) 
{     
        for (int j = 0; j < 3; j++) 
{ 
            printf("%d ", *ptr);
            ptr++;
        }
        printf("\n"); 
    }
}

Output
Enter 6 numbers:
10
20
30
40
50
60
The 2D array elements are:
10 20 30 
40 50 60

QUESTION 1:
Write a program to enter a number from the user. If the number is between 1 and 5, print the same number. Otherwise, print invalid number.

PROGRAM 1: Using Multiple If Statements

#include < stdio >

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    if (n == 1)
        printf("1");

    if (n == 2)
        printf("2");

    if (n == 3)
        printf("3");

    if (n == 4)
        printf("4");

    if (n == 5)
        printf("5");

    if (n < 1 || n > 5)
        printf("Invalid number");

    return 0;
}

OUTPUT:
Enter a number: 3
3

QUESTION 15. Search a value in an array. If found, print Found; otherwise, print Not Found.

    {
        scanf("%d", &a[i]);

            break;
        }

    }

}

QUESTION 16. Enter five values and sort the array in ascending order.

PROGRAM:

#include < stdio.h >
int main()
{
    int a[5], i, j, temp;
    printf("Enter 5 values: ");
    for(i=0; i<5; i++)
    {
        scanf("%d", &a[i]);
    }
    for(i=0; i<5; i++)
    {
        for(j=i+1; j<5; j++)
        {
            if(a[i]>a[j])
            {
                temp=a[i];
                a[i]=a[j];
                a[j]=temp;
            }
        }
    }
    printf("Sorted array: ");
    for(i=0; i<5; i++)
    {
        printf("%d ", a[i]);
    }
    return 0;
}

OUTPUT:
Enter 5 values: 50 20 40 10 30
Sorted array: 10 20 30 40 50

Question 1

Enter the four numbers from user and print it into matrix.

Solution:

#include  < stdio.h >;

int main()
{
    int num1, num2, num3, num4;

    printf("Enter 4 numbers: ");
    scanf("%d", &num1);
    scanf("%d", &num2);
    scanf("%d", &num3);
    scanf("%d", &num4);

    printf("\nMatrix:\n");
    printf("%d  %d\n", num1, num2);
    printf("%d  %d\n", num3, num4);

    return 0;
}

Question 2

Enter the matrix from user and transpose them.

Solution:

#include < stdio.h >;

int main()
{
    int matrix[2][2], i, j;

    printf("Enter 4 numbers for matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &matrix[i][j]);
        }
    }

    printf("\nOriginal Matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");
    }

    printf("\nTransposed Matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", matrix[j][i]);
        }
        printf("\n");
    }

    return 0;
}

Output:

Enter 4 numbers for matrix: 1 2 3 4 Original Matrix: 1 2 3 4 Transposed Matrix: 1 3 2 4

Question 4

Solution:

{

    int i, j;

    printf("Enter 4 numbers for Matrix A:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &A[i][j]);
        }
    }

    printf("Enter 4 numbers for Matrix B:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &B[i][j]);
        }
    }

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            C[i][j] = A[i][j] - B[i][j];
        }
    }

    printf("\nSubtraction:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", C[i][j]);
        }
        printf("\n");
    }

}

Question 5

Enter 2 matrix from user and multiply them.

Solution:

#include < stdio.h >;

int main()
{
    int A[2][2], B[2][2], C[2][2];
    int i, j, k;

    printf("Enter 4 elements for First Matrix (A):\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &A[i][j]);
        }
    }

    printf("Enter 4 elements for Second Matrix (B):\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &B[i][j]);
        }
    }

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            C[i][j] = 0;

            for(k = 0; k < 2; k++)
            {
                C[i][j] = C[i][j] + A[i][k] * B[k][j];
            }
        }
    }

    printf("\nResult Matrix (A * B):\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", C[i][j]);
        }
        printf("\n");
    }

    return 0;
}

Output:

Enter 4 elements for First Matrix (A): 1 2 3 4 Enter 4 elements for Second Matrix (B): 5 6 7 8 Result Matrix (A * B): 19 22 43 50

Question 6

Write a C program to input and display a 3 x 3 matrix using a two-dimensional array.

Solution:

#include < stdio.h >;

int main()
{
    int number[3][3];
    int i, j;

    for(i = 0; i < 3; i++)
    {
        for(j = 0; j < 3; j++)
        {
            printf("Enter number: ");
            scanf("%d", &number[i][j]);
        }
    }

    printf("The matrix is:\n");

    for(i = 0; i < 3; i++)
    {
        for(j = 0; j < 3; j++)
        {
            printf("%d\t", number[i][j]);
        }
        printf("\n");
    }

    return 0;
}

QUESTION:

Write a C program to input and display a 3 × 3 matrix using a two-dimensional array.
PROGRAM (Your notebook code, written neatly)
#include < stdio.h >

int main()
{
    int number[3][3];
    int i, j;

    for(i = 0; i < 3; i++)
    {
        for(j = 0; j < 3; j++)
        {
            printf("Enter number: ");
            scanf("%d", &number[i][j]);
        }
    }

    printf("The matrix is:\n");

    for(i = 0; i < 3; i++)
    {
        for(j = 0; j < 3; j++)
        {
            printf("%d\t", number[i][j]);
        }
        printf("\n");
    }

    return 0;
}
